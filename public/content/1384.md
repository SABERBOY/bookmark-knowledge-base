#Using torchvision.transforms.ToTensor  
to_tensor_tfm=tfms.ToTensor()  
  
def pil_to_latent(input_im):  
#Single image->single latent in a batch(so siz e 1,4,64,64)  
with torch.no_grad():  
latent=vae.encode(to_tensor_tfm(input_im).un squeeze(0).to(torch_device)*2-1)#Note scaling  
return 0.18215*latent.mode()#or.mean or.sample  
  
def latents_to_pil(latents):  
#bath of latents->list of images  
latents=(1/0.18215)*latents  
with torch.no_grad():  
image=vae.decode(latents)[0]  
image=(image/2+0.5).clamp(0,1)  
image=image.detach().cpu().permute(0,2,3,1).numpy()  
images=(image*255).round().astype("uint8")  
pil_images=[Image.fromarray(image)for image in images]  
return pil_images
